def clear():

def clean_input(text):
    if not text:
        return ""
    text = str(text).strip()
    text = ''.join(c for c in text if c.isprintable() and ord(c) >= 32)
    text = text.replace("^@", "").replace("^(", "").replace("�", "").replace("\x00", "")
    return text

def verify_token(token):
    headers = {"Authorization": token}
    try:
        r = requests.get("https://discord.com/api/v9/users/@me", headers=headers, timeout=10)
        if r.status_code == 200:
            user = r.json()
            print(f"{GREEN}✅ Valid Token → {user.get('username')}#{user.get('discriminator', '0000')}{RESET}")
            return True
        elif r.status_code == 401:
            print(f"{RED}❌ Invalid or expired token.{RESET}")
            return False
        else:
            print(f"{RED}❌ Error ({r.status_code}){RESET}")
            return False
    except:
        print(f"{RED}❌ Could not verify token.{RESET}")
        return False

def send_message(content, channel_id=None):
    if not channel_id:
        channel_id = CHANNEL_ID
    headers = {"Authorization": DISCORD_TOKEN, "Content-Type": "application/json"}
    try:
        r = requests.post(f"https://discord.com/api/v9/channels/{channel_id}/messages",
                         json={"content": content}, headers=headers, timeout=10)
        if r.status_code == 429:
            retry = r.json().get('retry_after', 1.5)
            print(f"{YELLOW}[!] Rate Limit → Waiting {retry}s{RESET}")
            time.sleep(retry)
            return False
        return r.status_code in [200, 204]
    except:
        return False

def send_to_webhook(webhook_url, content):
    try:
        payload = {"content": content}
        r = requests.post(webhook_url, json=payload, timeout=10)
        return r.status_code in [200, 204]
    except:
        return False

def generate_name(length=6):
    letters = string.ascii_lowercase
    numbers = string.digits
    if length <= 3:
        return ''.join(random.choices(letters, k=length))
    elif length == 4:
        return ''.join(random.choices(letters, k=4))
    elif length == 5:
        return ''.join(random.choices(letters, k=3)) + ''.join(random.choices(numbers, k=2))
    else:
        return ''.join(random.choices(letters + numbers, k=length))

def generate_gunslol_names(amount=20):
    prefixes = ["guns", "gun", "gs", "gunn", "gg", "xgun", "gunz"]
    suffixes = ["lol", "x", "69", "420", "77", "07", "yt", "op", "pro", "1337", "exe"]
    names = []
    for _ in range(amount):
        if random.random() > 0.5:
            name = random.choice(prefixes) + ''.join(random.choices(string.ascii_lowercase, k=random.randint(2,4))) + random.choice(suffixes)
        else:
            name = ''.join(random.choices(string.ascii_lowercase, k=random.randint(3,5))) + random.choice(suffixes)
        names.append(name)
    return names
