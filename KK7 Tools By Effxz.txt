def limpiar():
    print("\033[H\033[J", end="")

def limpiar_input(texto):
    if not texto: return ""
    texto = texto.strip()
    texto = ''.join(c for c in texto if c.isprintable())
    return texto.replace("^@", "").replace("^(", "").replace(" ", "")

def verificar_token(token):
    headers = {"Authorization": token}
    try:
        r = requests.get("https://discord.com/api/v9/users/@me", headers=headers, timeout=10)
        if r.status_code == 200:
            user = r.json()
            print(f"{VERDE}✅ Token válido → {user.get('username')}#{user.get('discriminator', '0000')}{RESET}")
            return True
        elif r.status_code == 401:
            print(f"{ROJO}❌ Token inválido o expirado.{RESET}")
            return False
        else:
            print(f"{ROJO}❌ Error ({r.status_code}){RESET}")
            return False
    except:
        print(f"{ROJO}❌ No se pudo verificar el token.{RESET}")
        return False

            print(f"   [{num}] {color}{nombre}{RESET}")
        print(f"\n   [0] Volver sin cambiar\n")
        op = input(f"{COLOR}{BOLD}   Elige un color → {RESET}").strip()
        if op == "0":
            return
        elif op in COLORES:
            COLOR = COLORES[op][0]
            print(f"\n{COLOR}✅ Color cambiado correctamente.{RESET}")
            time.sleep(1.3)
            return
        else:
            print(f"{ROJO}Opción inválida.{RESET}")
            time.sleep(1)

def enviar_mensaje(contenido, channel_id=None):
    if not channel_id:
        channel_id = CHANNEL_ID
    headers = {"Authorization": DISCORD_TOKEN, "Content-Type": "application/json"}
    try:
        r = requests.post(f"https://discord.com/api/v9/channels/{channel_id}/messages",
                         json={"content": contenido}, headers=headers, timeout=10)
        if r.status_code == 429:
            retry = r.json().get('retry_after', 1)
            print(f"{AMARILLO}[!] Rate Limit → Esperando {retry}s{RESET}")
            time.sleep(retry)
            return False
        return r.status_code in [200, 204]
    except:
        return False

def get_all_members():
    """Obtiene TODOS los miembros del servidor con paginación"""
    headers = {"Authorization": DISCORD_TOKEN}
    members = []
    after = None
    print(f"{AMARILLO}[!] Obteniendo lista completa de miembros...{RESET}")
    
    while True:
        url = f"https://discord.com/api/v9/guilds/{GUILD_ID}/members?limit=1000"
        if after:
            url += f"&after={after}"
        
        try:
            r = requests.get(url, headers=headers, timeout=10)
            if r.status_code != 200:
                print(f"{ROJO}Error al obtener miembros: {r.status_code}{RESET}")
                break
            
            data = r.json()
            if not data:
                break
                
            members.extend(data)
            after = data[-1]['user']['id'] if data else None
            
            print(f"{VERDE}→ Obtenidos {len(members)} miembros...{RESET}", end="\r")
            
            if len(data) < 1000:
                break
            time.sleep(0.3)  # Evitar rate limit
        except:
            break
    
    print(f"\n{VERDE}✅ Total miembros encontrados: {len(members)}{RESET}")
    return members
